from PIL import Image, ImageDraw import os os.makedirs('/home/claude/genou-pwa/icons', exist_ok=True) for size in [192, 512]: img = Image.new('RGB', (size, size), '#0c0c10') draw = ImageDraw.Draw(img) # Draw a simple knee icon margin = size // 6 cx, cy = size // 2, size // 2 r = size // 3 draw.ellipse([cx-r, cy-r, cx+r, cy+r], outline='#f59e0b', width=max(2, size//40)) # Cross/plus inside lw = max(2, size//50) draw.line([cx, cy-r//2, cx, cy+r//2], fill='#f59e0b', width=lw) draw.line([cx-r//2, cy, cx+r//2, cy], fill='#f59e0b', width=lw) img.save(f'/home/claude/genou-pwa/icons/icon-{size}.png') print(f"Generated {size}x{size}")
